cmake_minimum_required(VERSION 3.16)
project(mamma_512_staged_lm LANGUAGES CXX)
if(NOT IS_DIRECTORY "${PANO_CPP}" OR NOT EXISTS "${ADAPTED_SOURCE}")
  message(FATAL_ERROR "Set PANO_CPP and ADAPTED_SOURCE to existing read-only source and generated worker")
endif()
add_subdirectory("${PANO_CPP}" "${CMAKE_CURRENT_BINARY_DIR}/pano_staged_lm")
add_executable(mamma_512_worker "${ADAPTED_SOURCE}")
target_link_libraries(mamma_512_worker PRIVATE staged_lm)
target_compile_options(mamma_512_worker PRIVATE -ffp-contract=off)
